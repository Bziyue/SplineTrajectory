add_custom_target(examples)
foreach(name trajectory_sampling basic_optimizer corridor_optimizer esdf_optimizer dynamics_optimizer convex_hull_optimizer parameterized_optimizer)
    spline_executable(${name} ${name}.cpp)
    target_include_directories(${name} SYSTEM PRIVATE
        ${SplineTrajectory_SOURCE_DIR}/benchmarks/planners/upstream/GCOPTER/gcopter/include)
    add_dependencies(examples ${name})
    if(BUILD_TESTING)
        if(name STREQUAL "trajectory_sampling" OR name STREQUAL "parameterized_optimizer")
            add_test(NAME example_${name} COMMAND ${name})
        else()
            add_test(NAME example_${name} COMMAND ${name} --check-gradients)
        endif()
        set_tests_properties(example_${name} PROPERTIES LABELS "examples")
    endif()
endforeach()
