    SplineInput contains durations, waypoint rows and endpoint derivatives. spline.update(input) interpolates every waypoint. In optimization, the same values initialize the selected variables: the default mask allows durations and inner waypoints to change, while endpoints and boundary derivatives stay fixed.SplineInput 保存时长、途经点行和端点导数。spline.update(input) 插值所有途经点；用于优化时，同样的数据作为选中变量的初值。默认允许时长和内部途经点改变，端点位置与边界导数固定。

    Build a trajectory and sample position, velocity and acceleration构造轨迹并采样位置、速度、加速度C++17
    #include "SplineTrajectory.hpp"
#include <iostream>

int main() {
    using Spline = SplineTrajectory::QuinticSplineND<3>;
    SplineTrajectory::SplineInput<3> input;
    input.durations = {1.2, 1.5};
    input.waypoints.resize(3, 3);
    input.waypoints << 0, 0, 0,
                      1, 0.5, 0,
                      2, 0, 0;
    input.start_time = 2.0;
    // Endpoint velocity and acceleration default to zero.
    Spline spline;
    if (const auto status = spline.update(input); !status) {
        std::cerr << status.message << '\n';
        return 1;
    }

    auto trajectory = spline.copyPolynomial();
    auto sampler = trajectory.makeSampler<2>();
    for (int i = 0; i <= 10; ++i) {
        const double t = trajectory.startTime() + trajectory.duration() * i / 10.0;
        const auto pva = sampler.evaluateDerivatives(t);
        std::cout << t << " p: " << pva[0].transpose()
                  << " v: " << pva[1].transpose()
                  << " a: " << pva[2].transpose() << '\n';
    }
}

The two runnable examples below describe the same trajectory using durations or trajectory times.     Initialize from segment durations使用段时长初始化C++17
    #include "SplineTrajectory.hpp"
#include <iostream>

int main() {
    using Spline = SplineTrajectory::QuinticSplineND<3>;
    Spline::WaypointMatrix waypoints(3, 3);
    waypoints << 0, 0, 0,  1, 0.5, 0,  2, 0, 0;
    SplineTrajectory::BoundaryConditions<3> bc;
    bc.start.v << 0.2, 0, 0;

    #include "SplineTrajectory.hpp"
#include <iostream>

int main() {
    using Spline = SplineTrajectory::QuinticSplineND<3>;
    Spline::WaypointMatrix waypoints(3, 3);
    waypoints << 0, 0, 0,  1, 0.5, 0,  2, 0, 0;
    Spline spline;
    if (const auto status = spline.update({1.2, 1.5}, waypoints, 0.0); !status) return 1;
    const auto &poly = spline.polynomial();
    const std::vector<double> times{0.0, 0.5, 1.2, 2.7};

    auto pva = poly.evaluateDerivatives<2>(0.5);
    Spline::WaypointMatrix velocity(times.size(), 3);
    poly.evaluate(times, velocity, SplineTrajectory::Deriv::Vel);
    auto sampler = poly.makeSampler<2>();
    auto position = sampler.evaluate(0.5);
    auto batch = sampler.evaluateDerivatives(times);
    std::cout << position.transpose() << '\n' << batch[1] << '\n';
    return pva[1].isApprox(velocity.row(1).transpose()) ? 0 : 1;
}

    #include "SplineOptimizer.hpp"
#include <gcopter/lbfgs.hpp>
#include <iostream>
#include <numeric>

using Spline = SplineTrajectory::QuinticSplineND<3>;
using Optimizer = SplineTrajectory::SplineOptimizer<Spline>;

struct VelocityCost {
    static constexpr int kDerivativeOrder = 1;
    double weight = 0.02;

    double operator()(const SplineTrajectory::IntegralPointInfo&,
                      const SplineTrajectory::IntegralState<3>& state,
                      SplineTrajectory::IntegralGradient<3>& gradient) const {
        gradient.v += 2.0 * weight * state.v;
        return weight * state.v.squaredNorm();
    }
};

struct TargetDurationCost {
    double target = 5.0;
    double weight = 0.2;

    double operator()(const SplineTrajectory::ParameterView<Spline>& state,
                      SplineTrajectory::ParameterGradientView<Spline>& gradient) const {
        const double duration = std::accumulate(
            state.durations.begin(), state.durations.end(), 0.0);
        const double error = duration - target;
        gradient.durations.array() += 2.0 * weight * error;
        return weight * error * error;
    }
};

struct Objective {
    VelocityCost integral;
    TargetDurationCost parameter;
};

int main() {
    SplineTrajectory::SplineInput<3> input;
    input.durations = {1.2, 1.5};
    input.waypoints.resize(3, 3);
    input.waypoints << 0, 0, 0,  1, 0.5, 0,  2, 0, 0;
    SplineTrajectory::OptimizerOptions options;
    options.energy_weight = 0.02;
    options.duration_weight = 0.5;
    options.integration_steps = 32;
    Objective objective;
    Optimizer optimizer;
    if (const auto status = optimizer.prepare(input, options, objective); !status) {
        std::cerr << status.message << '\n';
        return 1;
    }

    auto x = optimizer.initialGuess();
    struct Context { Optimizer& optimizer; Objective& objective; } context{optimizer, objective};
    auto evaluate = [](void* instance, const Eigen::VectorXd& variables,
                       Eigen::VectorXd& gradient) {
        auto& ctx = *static_cast<Context*>(instance);
        return ctx.optimizer.evaluate(variables, gradient, ctx.objective).cost;
    };
    lbfgs::lbfgs_parameter_t parameters;
    parameters.g_epsilon = 1e-6;
    parameters.past = 5;
    parameters.delta = 1e-9;
    parameters.max_iterations = 500;
    double cost = 0.0;
    const int result = lbfgs::lbfgs_optimize(
        x, cost, evaluate, nullptr, nullptr, &context, parameters);
    if (result < 0) {
        std::cerr << lbfgs::lbfgs_strerror(result) << '\n';
        return 1;
    }
    if (const auto status = optimizer.update(x); !status) {
        std::cerr << status.message << '\n';
        return 1;
    }
    // Check application constraints here before accepting the trajectory.
    auto trajectory = optimizer.copyPolynomial();
    auto sampler = trajectory.makeSampler<2>();
    const auto pva = sampler.evaluateDerivatives(
        trajectory.startTime() + 0.5 * trajectory.duration());
    std::cout << "Final cost: " << cost << "\nPosition: " << pva[0].transpose()
              << "\nVelocity: " << pva[1].transpose() << '\n';
}

    struct VelocityCost {
    static constexpr int kDerivativeOrder = 1;
    double weight = 0.02;

    double operator()(const SplineTrajectory::IntegralPointInfo&,
                      const SplineTrajectory::IntegralState<3>& state,
                      SplineTrajectory::IntegralGradient<3>& gradient) const {
        gradient.v += 2.0 * weight * state.v;
        return weight * state.v.squaredNorm();
    }
};

struct TargetDurationCost {
    double target = 5.0;
    double weight = 0.2;

    double operator()(const SplineTrajectory::ParameterView<Spline>& state,
                      SplineTrajectory::ParameterGradientView<Spline>& gradient) const {
        const double duration = std::accumulate(
            state.durations.begin(), state.durations.end(), 0.0);
        const double error = duration - target;
        gradient.durations.array() += 2.0 * weight * error;
        return weight * error * error;
    }
};

struct Objective {
    VelocityCost integral;
    TargetDurationCost parameter;
};

    #include "SplineConvexHull.hpp"
#include <iostream>

int main() {
    using Spline = SplineTrajectory::QuinticSplineND<3>;
    Spline::WaypointMatrix waypoints(3, 3);
    waypoints << 0, 0, 0,  1, 0.5, 0,  2, 0, 0;
    Spline spline;
    if (const auto status = spline.update({1.2, 1.5}, waypoints, 0.0); !status) return 1;
    SplineTrajectory::ConvexHullWorkspace<3> hull;
    hull.prepare(spline.polynomial(), {SplineTrajectory::ConvexHullBasis::Bezier, 0, 1});
    hull.update(spline.polynomial());

    // L = 0.5 * ||controls||^2, dL/dControls = controls.
    auto dControls = hull.controls().eval();
    auto partials = hull.backward(dControls);
    auto gradients = spline.backward(partials.coefficients, partials.durations);
    std::cout << hull.numPieces() << '\n' << gradients.durations.transpose() << '\n';
}
