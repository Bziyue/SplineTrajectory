    Both update forms reuse fixed-topology storage and recompute coefficients/factors. Failure invalidates the polynomial but retains workspace for retry; any update invalidates borrowed views.     Initialize from segment durations使用段时长初始化C++17
    #include "SplineTrajectory.hpp"
#include <iostream>

int main() {
    using Spline = SplineTrajectory::QuinticSplineND<3>;
    Spline::CoefficientMatrix points(3, 3);
    points << 0, 0, 0,  1, 0.5, 0,  2, 0, 0;
    SplineTrajectory::BoundaryConditions<3> bc;
    bc.start.v << 0.2, 0, 0;

    const std::vector<double> durations{1.2, 1.5};
    Spline spline;
    const auto status = spline.update(durations, points, 0.0, bc);
    if (!status) { std::cerr << status.message; return 1; }
    std::cout << spline.polynomial().evaluate(0.5).transpose() << '\n';
}

    Runnable sampling example可运行的采样示例C++17
    #include "SplineTrajectory.hpp"
#include <iostream>

int main() {
    using Spline = SplineTrajectory::QuinticSplineND<3>;
    Spline::CoefficientMatrix points(3, 3);
    points << 0, 0, 0,  1, 0.5, 0,  2, 0, 0;
    Spline spline;
    if (const auto status = spline.update({1.2, 1.5}, points, 0.0); !status) return 1;
    const auto &poly = spline.polynomial();
    const std::vector<double> times{0.0, 0.5, 1.2, 2.7};

    auto pva = poly.evaluateDerivatives<2>(0.5);
    Spline::CoefficientMatrix velocity(times.size(), 3);
    poly.evaluate(times, velocity, SplineTrajectory::Deriv::Vel);
    auto sampler = poly.makeSampler<2>();
    auto position = sampler.evaluate(0.5);
    auto batch = sampler.evaluateDerivatives(times);
    std::cout << position.transpose() << '\n' << batch[1] << '\n';
    return pva[1].isApprox(velocity.row(1).transpose()) ? 0 : 1;
}

    Runnable optimizer evaluation (no solver dependency)可运行的优化器求值示例（不依赖求解器）C++17
    #include "SplineOptimizer.hpp"
#include <iostream>

int main() {
    using Spline = SplineTrajectory::QuinticSplineND<3>;
    SplineTrajectory::SplineInput<3> problem;
    problem.durations = {1.2, 1.5};
    problem.waypoints.resize(3, 3);
    problem.waypoints << 0, 0, 0,  1, 0.5, 0,  2, 0, 0;
    SplineTrajectory::OptimizerOptions options;
    options.energy_weight = 1.0;
    SplineTrajectory::NoObjective objective;
    SplineTrajectory::SplineOptimizer<Spline> optimizer;
    auto status = optimizer.prepare(problem, options, objective);
    if (!status) return 1;

    auto x = optimizer.initialGuess();
    Eigen::VectorXd gradient(x.size());
    auto result = optimizer.evaluate(x, gradient, objective);
    if (!result) { std::cerr << result.status.message; return 1; }
    // An external solver repeats evaluate() and chooses an accepted x.
    if (!optimizer.update(x)) return 1;
    // Check planning constraints before accepting this candidate.
    auto trajectory = optimizer.copyPolynomial();
    std::cout << result.cost << '\n' << trajectory.evaluate(0.5).transpose() << '\n';
}
